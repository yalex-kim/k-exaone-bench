```js
function spokesMarkup(cx, cy, rIn, rOut, count, color){
```

```js
function pedalPos(angle){
  return { x: BB.x + CRANK_R * Math.cos(angle), y: BB.y - CRANK_R * Math.sin(angle) };
}
function legPath(hip, foot){
  const dx = foot.x - hip.x, dy = foot.y - hip.y;
  const len = Math.hypot(dx, dy) || 1;
  const bend = Math.min(22, len * 0.22);
  const kx = (hip.x + foot.x) / 2 + (dy / len) * bend;
  const ky = (hip.y + foot.y) / 2 - (dx / len) * bend;
  return `M ${f(hip.x)} ${f(hip.y)} Q ${f(kx)} ${f(ky)} ${f(foot.x)} ${f(foot.y)}`;
}
function setLeg(outEl, colEl, footEl, hip, foot){
  const d = legPath(hip, foot);
  outEl.setAttribute('d', d); colEl.setAttribute('d', d);
  footEl.setAttribute('cx', f(foot.x)); footEl.setAttribute('cy', f(foot.y));
}
```

frame loop:
```js
function frame(ts){
  if (!lastTs) lastTs = ts;
  const dt = Math.min(Math.max((ts - lastTs) / 1000, 0), 0.05);
  lastTs = ts;
  if (refs){
    if (!reducedMotion){
      driftClouds(dt);
      const omega = state.riding ? (state.speed / 100) * MAX_OMEGA : 0;
      if (omega > 0){
        state.angle = (state.angle + omega * dt) % TAU;
        applyPedal(state.angle);
      }
    }
  }
  requestAnimationFrame(frame);
}
```
